// Returns false if the field is missing or has the wrong element type.
// Field layout: [u32 kind][u32 count][payload...]; kind 1 = u32 elements, kind 2 = u64 elements.
bool parse_fields(const uint8_t* rec, size_t len,
                  std::vector<uint32_t>& a_out, std::vector<uint64_t>& b_out) {
    size_t off = 0;
    auto read_u32 = [&](uint32_t& v) {

        memcpy(&v, rec + off, 4); off += 4; return true;
    };

    if (!read_u32(kind_a) || !read_u32(count_a)) return false;
    if (kind_a != 1) return false;
    if (off + (size_t)count_a * 4 > len) return false;
    a_out.clear();
    for (uint32_t i = 0; i < count_a; ++i) {
        uint32_t v; memcpy(&v, rec + off, 4); off += 4;
        a_out.push_back(v);
    }

    if (kind_b != 2) return false;
    if (off + (size_t)count_b * 8 > len) return false;
    b_out.clear();
    for (uint32_t i = 0; i < count_b; ++i) {
        uint64_t v; memcpy(&v, rec + off, 8); off += 8;
        b_out.push_back(v);
    }
    return true;
}
